# 存档数据结构设计（PlayerState JSON Schema）

## 一、版本与迁移策略

```typescript
/** 存档版本号，每次结构变更递增 */
const SAVE_VERSION = 1;

// 注意：本结构中引用的 recipeId（配方ID）和 templateId（装备模板ID）
// 需要与游戏配置表对接。配置表schema将在开发阶段定义。
// 预期配置表：
// - RECIPES: Record<string, RecipeConfig> — 配方配置
// - EQUIPMENT_TEMPLATES: Record<string, EquipmentTemplate> — 装备模板配置
// - MONSTER_TEMPLATES: Record<string, MonsterTemplate> — 怪物模板配置
// - SKILL_TEMPLATES: Record<string, SkillTemplate> — 技能模板配置

/**
 * 版本迁移策略
 *
 * 1. 每次存档写入时附带当前 SAVE_VERSION
 * 2. 读取时检测版本号，逐级执行迁移函数
 * 3. 迁移链：v0 → migrateV0ToV1 → migrateV1ToV2 → ...
 * 4. 每个迁移函数只负责"上一版 → 本版"的差异补全，不做全量覆盖
 * 5. 新增字段必须提供默认值，删除字段在迁移中清理
 * 6. 迁移失败时保留原存档副本（key: backup_<timestamp>），防止数据丢失
 *
 * 迁移示例：
 *   假设 v2 新增"灵田浇水"字段 → migrateV1ToV2 在每个灵田 slot 上补 { watered: false }
 */
```

## 二、枚举与基础类型

```typescript
// ==================== 境界 ====================

/** 境界名 */
type RealmName = '练气' | '筑基' | '金丹' | '元婴' | '化神' | '渡劫' | '无限';

/** 境界进度（练气1阶 = { realm: '练气', tier: 1, level: 1 }） */
interface RealmProgress {
  /** 当前境界名 */
  realm: RealmName;
  /** 当前境界内的阶（1-10，无限境界无上限） */
  tier: number;
  /** 全局等级（练气1阶=1，渡劫10阶=60，无限境界 > 60） */
  level: number;
  /** 当前积累的星元（用于突破） */
  starYuan: number;
  /** 升到下一级所需星元 */
  starYuanToNext: number;
}

// ==================== 装备 ====================

/** 装备部位 */
type EquipmentSlot = 'weapon' | 'helmet' | 'armor' | 'gauntlet' | 'boot' | 'talisman';

/** 装备品质 */
type EquipmentQuality =
  | 'common'      // 普通（白）
  | 'fine'        // 精良（绿）
  | 'rare'        // 稀有（蓝）
  | 'epic'        // 史诗（紫）
  | 'legendary'   // 传说（橙）
  | 'mythic'      // 神话（红）
  | 'ancient';    // 太古（金，三周目解锁）

/** 词条类型 */
type AffixType =
  | 'atkPct'          // 攻击 +X%
  | 'defPct'          // 防御 +X%
  | 'hpPct'           // 生命 +X%
  | 'critRate'        // 暴击率 +X%
  | 'critDmg'         // 暴击伤害 +X%
  | 'penetration'     // 穿透 +X%
  | 'dodge'           // 闪避率 +X%
  | 'starYuanGain'    // 星元获取 +X%
  | 'gatherSpeed'     // 采集速度 +X%
  | 'prodSpeed'       // 生产速度 +X%
  | 'dropRate'        // 掉率提升 +X%
  | 'cdReduction'     // 技能冷却 -X%
  | 'skillDmg'        // 技能伤害 +X%
  | 'lifeSteal'       // 攻击吸血（特殊词条）
  | 'dmgReduction'    // 减伤（特殊词条）
  | 'allStats'        // 全属性（特殊词条）
  | 'berserkerRage'   // 狂战士之怒
  | 'ironWall'        // 铁壁守护
  | 'lifeDrain'       // 生命汲取
  | 'critMaster'      // 暴击大师
  | 'gatherMaster'    // 采集达人
  | 'prodMaster'      // 生产大师
  | 'luckyStar'       // 幸运之星
  | 'endlessPower';   // 无尽之力

/** 词条（槽位上的单条属性） */
interface Affix {
  /** 词条类型 */
  type: AffixType;
  /** 数值（百分比词条存小数，如 3% = 0.03） */
  value: number;
  /** 是否为负面词条 */
  isNegative: boolean;
}

/** 单件装备 */
interface Equipment {
  /** 装备唯一 ID（UUID） */
  id: string;
  /** 装备模板 ID（用于查找基础配置） */
  templateId: string;
  /** 装备名称 */
  name: string;
  /** 装备部位 */
  slot: EquipmentSlot;
  /** 品质 */
  quality: EquipmentQuality;
  /** 品质等级（无尽模式可突破品质上限） */
  qualityLevel: number;
  /** 强化等级（0 = 未强化，无上限） */
  enhanceLevel: number;
  /** 词条列表 */
  affixes: Affix[];
  /** 词条槽总数（基础5 + 每+10强化等级额外1个） */
  totalAffixSlots: number;
  /** 装备来源星球（用于套装判定） */
  sourcePlanet?: string;
  /** 是否已锁定（防止误分解） */
  locked: boolean;
  /** 获取时间戳 */
  acquiredAt: number;
}

// ==================== 功法 ====================

/** 功法类型 */
type SkillType = 'active' | 'passive';

/** 功法状态 */
type SkillStatus = 'locked' | 'available' | 'learned';

/** 功法定义 */
interface Skill {
  /** 功法 ID */
  id: string;
  /** 功法名称 */
  name: string;
  /** 类型：主动 / 被动 */
  type: SkillType;
  /** 状态 */
  status: SkillStatus;
  /** 进化等级（0 = 未进化，二周目解锁） */
  evolutionLevel: number;
  /** 功法残页已收集数量（需要残页合成的功法） */
  fragmentCount: number;
  /** 合成所需残页总数（0 = 不需要残页） */
  fragmentRequired: number;
  /** 冷却结束时间戳（毫秒，0 = 未在冷却中） */
  cooldownEndsAt: number;
}

// ==================== 生产系统 ====================

/** 设备类型 */
type DeviceType =
  | 'smelter'          // 熔炼炉
  | 'distiller'        // 蒸馏器
  | 'extractor'        // 萃取机
  | 'alchemyFurnace'   // 炼丹炉
  | 'refiner'          // 精炼台
  | 'forge'            // 锻造台
  | 'spaceFolder';     // 空间折叠器

/** 生产队列项 */
interface ProductionItem {
  /** 配方 ID */
  recipeId: string;
  /** 产物名称 */
  outputName: string;
  /** 生产数量 */
  quantity: number;
  /** 开始时间戳（毫秒） */
  startedAt: number;
  /** 预计完成时间戳（毫秒） */
  finishAt: number;
}

/** 单个生产设备 */
interface Device {
  /** 设备类型 */
  type: DeviceType;
  /** 是否已解锁（false = 未解锁，不可使用；true = 已解锁） */
  unlocked: boolean;
  /** 设备等级（0 = 最低级；未解锁时也为0，需结合 unlocked 字段判断状态）
   *  - unlocked: false + level: 0 = 未解锁
   *  - unlocked: true  + level: 0 = 已解锁最低级
   *  - unlocked: true  + level: N = 已升级到N级
   */
  level: number;
  /** 最大等级 */
  maxLevel: number;
  /** 生产队列 */
  queue: ProductionItem[];
  /** 队列容量（默认3，可升级） */
  queueCapacity: number;
}

// ==================== 星球系统 ====================

/** 星球 ID */
type PlanetId = 'origin' | 'xuantie' | 'lingzhi' | 'youming' | 'tianjing' | 'huoshan' | 'bingfeng' | 'hundun' | 'fukong' | 'shenyuan';

/** 灵田状态 */
interface FarmPlot {
  /** 灵田编号（1-12） */
  plotId: number;
  /** 当前种植的种子 ID（null = 空闲） */
  seedId: string | null;
  /** 种植时间戳 */
  plantedAt: number;
  /** 预计成熟时间戳 */
  harvestAt: number;
  /** 灵田等级（影响产量） */
  level: number;
  /** 是否自动补种（收获后自动种植相同种子） */
  autoReplant: boolean;
}

// ==================== 星球地图 ====================

/** 地图格子类型 */
type TileType =
  | 'empty'            // 空地（可建造/可探索）
  | 'resource'         // 资源点（采集目标）
  | 'obstacle'         // 障碍物（需清除后变为empty）
  | 'defenseSlot'      // 防御建筑槽位（可放置防御塔）
  | 'bossGate'         // Boss入口（通往Boss战的路径）
  | 'invasionPath';    // 入侵路径（怪物行进路线）

/** 地图格子 */
interface MapTile {
  /** 格子坐标 x（列） */
  x: number;
  /** 格子坐标 y（行） */
  y: number;
  /** 格子类型 */
  type: TileType;
  /** 资源点关联的资源ID（仅 type='resource' 时有效） */
  resourceId?: string;
  /** 障碍物清除所需消耗（仅 type='obstacle' 时有效，key=物品ID, value=数量） */
  clearCost?: Record<string, number>;
  /** 是否已探索/已解锁 */
  revealed: boolean;
}

/** 防御建筑类型（与11-数值平衡总表10.1节一致） */
type DefenseBuildingType =
  | 'arrowTower'       // 箭塔（物理，单体远程）
  | 'fireTower'        // 火焰塔（元素，灼烧DoT）
  | 'freezeTower'      // 冰冻塔（元素，减速控制）
  | 'lightningTower'   // 雷电塔（元素，链式弹射）
  | 'wall'             // 围墙（阻挡怪物前进）
  | 'trap';            // 陷阱（一次性高额伤害）

/** 防御建筑 */
interface DefenseBuilding {
  /** 建筑唯一 ID */
  id: string;
  /** 建筑类型 */
  type: DefenseBuildingType;
  /** 所在格子坐标 */
  tileX: number;
  tileY: number;
  /** 建筑等级（1-5，与11-数值平衡总表10.1节一致） */
  level: number;
  /** 当前耐久度 */
  durability: number;
  /** 最大耐久度 */
  maxDurability: number;
  /** 上次攻击时间戳（用于计算冷却） */
  lastAttackAt: number;
}

/** 入侵怪物 */
interface InvasionMonster {
  /** 怪物实例 ID */
  id: string;
  /** 怪物模板 ID（关联配置表） */
  templateId: string;
  /** 怪物名称 */
  name: string;
  /** 当前生命值 */
  currentHp: number;
  /** 最大生命值 */
  maxHp: number;
  /** 移动速度（格/秒） */
  moveSpeed: number;
  /** 当前所在格子坐标 */
  tileX: number;
  tileY: number;
  /** 行进路径索引（当前走到路径的第几格） */
  pathIndex: number;
  /** 是否为精英怪 */
  isElite: boolean;
  /** 掉落物（击杀后获得，key=物品ID, value=数量） */
  drops: Record<string, number>;
}

/** 入侵状态 */
interface InvasionState {
  /** 是否有入侵正在进行 */
  active: boolean;
  /** 当前入侵波次（从1开始） */
  wave: number;
  /** 总波次 */
  totalWaves: number;
  /** 当前波次剩余怪物生成队列 */
  pendingMonsters: { templateId: string; spawnDelay: number }[];
  /** 已出现在地图上的怪物列表 */
  activeMonsters: InvasionMonster[];
  /** 本波次已击杀怪物数 */
  killedCount: number;
  /** 本波次总怪物数 */
  totalCount: number;
  /** 入侵开始时间戳 */
  startedAt: number;
  /** 波次间隔（下一波开始前的等待秒数） */
  waveInterval: number;
  /** 下一波开始时间戳（0 = 当前波未结束） */
  nextWaveAt: number;
  /** 累计成功防御入侵次数 */
  defenseSuccessCount: number;
}

/** 星球状态 */
interface Planet {
  /** 星球 ID */
  id: PlanetId;
  /** 星球名称 */
  name: string;
  /** 是否已解锁 */
  unlocked: boolean;
  /** 自动采集是否开启 */
  autoCollect: boolean;
  /** 灵田最大格数（灵植星初始0，任务解锁后逐步增加到12；其他星球始终0） */
  maxFarmPlots: number;
  /** 灵田列表（仅灵植星有） */
  farmPlots: FarmPlot[];
  /** 已探索的区域 ID 列表 */
  exploredAreas: string[];
  /** Boss 上次击杀时间戳（用于计算刷新） */
  bossLastKilledAt: number;
  /** 隐藏 Boss 是否已解锁（二周目） */
  hiddenBossUnlocked: boolean;
  /** 星球地图格子（二维网格，y为行、x为列；防御战/入侵相关） */
  mapTiles: MapTile[];
  /** 已建造的防御建筑列表 */
  defenseBuildings: DefenseBuilding[];
  /** 入侵状态（防御战进行中的数据） */
  invasion: InvasionState;
  /** 隐藏Boss触发条件进度（key = 触发条件ID，value = 当前进度值）
   *  示例：
   *    xuantie: { consecutiveNoDeathKills: 7 }  // 连续无死亡击杀矿脉巨兽进度
   *    lingzhi:  { farmPlotsFilled: 9 }          // 灵田种满进度
   *    youming:  { ghostFireCollected: 65 }      // 幽冥火收集进度
   *    tianjing: { thunderstormBossKill: 0 }     // 雷暴天气击杀Boss标记（0/1）
   *    hundun:   { daoYunFragmentCollected: 30 } // 道韵碎片收集进度
   */
  hiddenBossProgress: Record<string, number>;
  /** 当前天气状态 */
  currentWeather: 'normal' | 'rain' | 'thunderstorm' | 'spatialRift';
  /** 天气剩余时间（秒） */
  weatherRemaining: number;
  /** 资源储量（key = 资源ID，如 'iron_ore'、'spirit_liquid'） */
  resourceReserves: Record<string, {
    /** 当前剩余储量 */
    current: number;
    /** 最大储量 */
    max: number;
    /** 每小时再生量，0=不可再生 */
    regenRate: number;
  }>;
}

// ==================== 离线系统 ====================
```

```typescript
 * 保存存档到 localStorage
 * - 自动更新 lastSavedAt
 * - 保存前做 JSON 序列化
 * - 保留最近 3 个备份
 */
function saveToLocalStorage(state: PlayerState): void {
  try {
    state.lastSavedAt = Date.now();
    const serialized = JSON.stringify(state);

    // 保存主存档
    localStorage.setItem(SAVE_KEY, serialized);

    // 保留最近 3 个备份
    const backupKey = `${BACKUP_PREFIX}${state.lastSavedAt}`;
    localStorage.setItem(backupKey, serialized);
    pruneOldBackups(3);
  } catch (e) {
    console.error('[存档] 保存失败:', e);
  }

/**
 * 从 localStorage 加载存档
 * - 自动执行版本迁移
 * - 加载失败返回 null
 */
function loadFromLocalStorage(): PlayerState | null {
  try {
    const raw = localStorage.getItem(SAVE_KEY);
    if (!raw) return null;

    const parsed = JSON.parse(raw) as Partial<PlayerState>;

    // 版本迁移
    const migrated = migrateIfNeeded(parsed);

    // 数据完整性校验
    if (!validatePlayerState(migrated)) {
      console.error('[存档] 数据校验失败，尝试从备份恢复');
      return loadFromBackup();
    }

    return migrated;
  } catch (e) {
    console.error('[存档] 加载失败:', e);
    return loadFromBackup();
  }
}

/**
 * 简单深合并（target 为默认值，source 为已有数据，source 优先）
 */
function deepMerge(target: any, source: any): any {
  if (source === null || source === undefined) return target;
  if (typeof source !== 'object' || typeof target !== 'object') return source;
  if (Array.isArray(source)) return source;

  const result: any = { ...target };
  for (const key of Object.keys(source)) {
    if (key in target && typeof target[key] === 'object' && !Array.isArray(target[key])) {
      result[key] = deepMerge(target[key], source[key]);
    } else {
      result[key] = source[key];
    }
  }
  return result;
}

/**
 * 数据完整性校验
 * - 检查关键字段是否存在
 * - 检查数值是否在合理范围
 */
function validatePlayerState(state: any): state is PlayerState {
  if (!state || typeof state !== 'object') return false;
  if (typeof state.version !== 'number') return false;
  if (!state.realm || typeof state.realm.level !== 'number') return false;
  if (!state.currency) return false;
  if (!state.planets) return false;
  if (state.realm.level < 1) return false;
  if (state.currency.starCoins < 0) return false;
  return true;
}

/**
 * 从备份恢复最近的存档
 */
function loadFromBackup(): PlayerState | null {
  const backups: { key: string; timestamp: number }[] = [];

  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key?.startsWith(BACKUP_PREFIX)) {
      const ts = parseInt(key.replace(BACKUP_PREFIX, ''), 10);
      if (!isNaN(ts)) backups.push({ key, timestamp: ts });
    }
  }

  if (backups.length === 0) return null;

  // 按时间戳降序，取最新的
  backups.sort((a, b) => b.timestamp - a.timestamp);

  for (const backup of backups) {
    try {
      const raw = localStorage.getItem(backup.key);
      if (!raw) continue;
      const parsed = JSON.parse(raw);
      if (validatePlayerState(parsed)) {
        console.warn(`[存档] 从备份 ${backup.key} 恢复成功`);
        return migrateIfNeeded(parsed);
      }
    } catch { /* skip invalid backup */ }
  }

  return null;
}

/**
 * 清理旧备份，保留最近 N 个
 */
function pruneOldBackups(keepCount: number): void {
  const backups: { key: string; timestamp: number }[] = [];

  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key?.startsWith(BACKUP_PREFIX)) {
      const ts = parseInt(key.replace(BACKUP_PREFIX, ''), 10);
      if (!isNaN(ts)) backups.push({ key, timestamp: ts });
    }
  }

  if (backups.length <= keepCount) return;

  backups.sort((a, b) => b.timestamp - a.timestamp);
  for (let i = keepCount; i < backups.length; i++) {
    localStorage.removeItem(backups[i].key);
  }
}

/**
 * 删除存档（含所有备份）
 */
function deleteSave(): void {
  localStorage.removeItem(SAVE_KEY);
  const keysToRemove: string[] = [];
  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key?.startsWith(BACKUP_PREFIX)) keysToRemove.push(key);
  }
  keysToRemove.forEach(k => localStorage.removeItem(k));
}

/**
 * 导出存档为 JSON 字符串（用于分享/手动备份）

function exportSave(state: PlayerState): string {
  return JSON.stringify(state, null, 2);
}

 */
function importSave(json: string): PlayerState | null {
  try {
    const parsed = JSON.parse(json);
    const migrated = migrateIfNeeded(parsed);
    if (!validatePlayerState(migrated)) return null;
    return migrated;
  } catch {
    return null;
  }
}
```

```typescript
/**
 * 计算离线收益
 *
 * 逻辑：
 * 1. 计算离线时长（受限于 maxOfflineSeconds）
 * 2. 遍历已开启自动采集的星球，按资源产出速率 × 离线时长计算收益
 * 3. 生产队列中的项目如果在离线期间完成，自动产出
 * 4. 灵田如果在离线期间成熟，自动收获
 * 5. 所有收益乘以 offlineMultiplier
 * 6. 返回收益快照（不立即写入 state，由调用方确认后写入）
 *
 * @param state 当前玩家状态
 * @returns 离线收益快照
 */
function calculateOfflineRewards(state: PlayerState): OfflineRewards {
  const now = Date.now();
  const offlineMs = now - state.lastOnlineAt;
  const offlineSeconds = Math.min(
    Math.floor(offlineMs / 1000),
    state.offlineConfig.maxOfflineSeconds
  );

  if (offlineSeconds < 60) {
    // 离线不足1分钟，不计算收益
    return {
      offlineSeconds: 0,
      resources: {},
      starCoins: 0,
      starYuan: 0,
      farmHarvests: {},
      claimed: true,
    };
  }

  const multiplier = state.offlineConfig.offlineMultiplier;
  const minutes = offlineSeconds / 60;
  const resources: Record<string, number> = {};
  let starCoins = 0;
  let starYuan = 0;

  // ---------- 1. 自动采集收益 ----------
  for (const planetId of state.offlineConfig.autoCollectPlanets) {
    const planet = state.planets[planetId];
    if (!planet?.unlocked || !planet.autoCollect) continue;

    const rates = getAutoCollectRates(planetId);
    for (const [resourceId, ratePerMin] of Object.entries(rates)) {
      const amount = Math.floor(ratePerMin * minutes * multiplier);
      if (amount > 0) {
        resources[resourceId] = (resources[resourceId] ?? 0) + amount;
      }
    }
  }

  // ---------- 2. 生产队列完成 ----------
  if (state.offlineConfig.autoProduce) {
    for (const [deviceType, device] of Object.entries(state.devices)) {
      if (device.level === 0) continue;

      for (const item of device.queue) {
        if (item.finishAt <= now) {
          // 生产完成
          resources[item.outputName] = (resources[item.outputName] ?? 0) + item.quantity;
        }
      }
    }
  }

  // ---------- 3. 灵田收获 ----------
  if (state.offlineConfig.autoHarvest) {
    const lingzhi = state.planets.lingzhi;
    if (lingzhi?.unlocked) {
      for (const plot of lingzhi.farmPlots) {
        if (!plot.seedId) continue;

        // 计算离线期间成熟了几次
        const cycleMs = plot.harvestAt - plot.plantedAt;
        if (cycleMs <= 0) continue;

        const elapsedMs = now - plot.plantedAt;
        const harvestCount = Math.floor(elapsedMs / cycleMs);

        if (harvestCount > 0) {
          const output = getSeedOutput(plot.seedId, plot.level);
          const totalAmount = Math.floor(output.amount * harvestCount * multiplier);
          resources[output.name] = (resources[output.name] ?? 0) + totalAmount;
        }
      }
    }
  }

  // ---------- 4. 境界星元收益 ----------
  // 离线期间按当前境界的星元获取效率计算，乘以周目星元倍率
  const starYuanPerMin = getStarYuanPerMinute(state);
  starYuan = Math.floor(starYuanPerMin * minutes * multiplier * state.cycle.starYuanMultiplier);

  // ---------- 5. 星币离线收益（按当前境界估算，乘以周目掉落数量倍率） ----------
  const starCoinsPerMin = getStarCoinsPerMinute(state);
  starCoins = Math.floor(starCoinsPerMin * minutes * multiplier * state.cycle.dropMultiplier);

  return {
    offlineSeconds,
    resources,
    starCoins,
    starYuan,
    farmHarvests: {},
    claimed: false,
  };
}

/**
 * 应用离线收益到玩家状态
 */
function applyOfflineRewards(state: PlayerState, rewards: OfflineRewards): void {
  if (rewards.claimed || rewards.offlineSeconds === 0) return;

  // 添加货币
  state.currency.starCoins += rewards.starCoins;
  state.currency.spiritStones += 0; // 灵石不通过离线自动获取
  state.realm.starYuan += rewards.starYuan;

  // 添加资源到背包
  for (const [resourceId, amount] of Object.entries(rewards.resources)) {
    addInventoryItem(state, resourceId, amount);
  }

  // 标记已领取
  rewards.claimed = true;
  state.pendingOfflineRewards = null;

  // 更新统计
  state.stats_log.totalOfflineSeconds += rewards.offlineSeconds;
  state.stats_log.totalStarCoinsEarned += rewards.starCoins;
}

/**
 * 获取种子产出信息
 */
function getSeedOutput(seedId: string, farmLevel: number): { name: string; amount: number } {
  const seedOutputs: Record<string, { name: string; baseAmount: number }> = {
    spirit_grass_seed:       { name: '灵草', baseAmount: 4 },
    centennial_grass_seed:   { name: '百年灵草', baseAmount: 3 },
    millennial_grass_seed:   { name: '千年灵草', baseAmount: 2 },
  };
  const seed = seedOutputs[seedId];
  if (!seed) return { name: 'unknown', amount: 0 };

  // 灵田等级加成：每级 +10% 产量
  const levelBonus = 1 + farmLevel * 0.1;
  return { name: seed.name, amount: Math.floor(seed.baseAmount * levelBonus) };
}

/**
 * 获取当前境界每分钟星元获取速率
 */
function getStarYuanPerMinute(state: PlayerState): number {
  // 简化公式：基于境界等级的基准速率
  const base = 10 + state.realm.level * 5;
  // 可叠加功法/成就加成
  return base;
}

/**
 * 获取当前境界每分钟星币获取速率
 */
function getStarCoinsPerMinute(state: PlayerState): number {
  // 基于境界等级的基准速率
  const base = 50 + state.realm.level * 30;
  return base;
}

/**
 * 向背包添加物品
 */
function addInventoryItem(state: PlayerState, itemId: string, amount: number): void {
  const existing = state.inventory.items[itemId];
  if (existing) {
    existing.quantity += amount;
  } else {
    state.inventory.items[itemId] = {
      id: itemId,
      name: itemId, // 实际应从配置表查询名称
      type: 'resource',
      quantity: amount,
      maxStack: 0,
    };
  }
}
```

| 稀疏存储 | 默认值的字段不写入 JSON（如未解锁的星球不存 farmPlots） |
| ID 映射 | 物品/功法用字符串 ID 引用配置表，不存完整属性 |
| 批量清理 | 已完成且超过 7 天的任务/成就只保留 ID，不保留详情 |

### 6.2 自动保存时机

| 事件 | 触发保存 |
|------|---------|
| 切换星球 | 是 |
| 完成任务 | 是 |
| 装备变更 | 是 |
| 境界突破 | 是 |
| 每 30 秒 | 定时保存 |
| 页面关闭 / 切后台 | beforeunload / visibilitychange |
| 购买商品 | 是 |

### 6.3 防篡改

| 措施 | 说明 |
|------|------|
| 校验和 | 存档附带 CRC32 或 SHA-256 摘要，加载时校验 |
| 关键数值上界 | 货币/属性超出合理范围时标记异常 |
| 存档加密 | 本地存储时用简单 XOR 或 AES 加密（防手改 JSON） |
